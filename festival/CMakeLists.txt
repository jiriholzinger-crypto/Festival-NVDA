# CMake build of the Festival library (libFestival) for Windows x64.
# Works with MSVC (x64) and MinGW-w64 / llvm-mingw.
# Requires Edinburgh Speech Tools built beforehand (see festlib64/BUILD.md).
cmake_minimum_required(VERSION 3.16)
project(festival CXX C)

set(EST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../speech_tools" CACHE PATH "speech_tools source dir")

set(MODULES Lexicon base Duration Intonation Text UniSyn donovan parser
            UniSyn_diphone clunits clustergen MultiSyn hts_engine)

file(GLOB ARCH_SRCS src/arch/festival/*.cc)
set(FEST_SRCS ${ARCH_SRCS})
foreach(m ${MODULES})
  file(GLOB m_cc src/modules/${m}/*.cc)
  file(GLOB m_c  src/modules/${m}/*.c)
  list(APPEND FEST_SRCS ${m_cc} ${m_c})
endforeach()
file(GLOB TMPL_SRCS src/modules/MultiSyn/inst_tmpl/*.cc)
list(APPEND FEST_SRCS ${TMPL_SRCS})
# explicit template instantiation files (CXX_COMMAND_TEMPLATES in the Makefiles)
set_source_files_properties(${TMPL_SRCS} PROPERTIES COMPILE_DEFINITIONS INSTANTIATE_TEMPLATES)

# init_modules.cc (normally generated by src/modules/Makefile)
set(INIT_MODULES "${CMAKE_CURRENT_BINARY_DIR}/init_modules.cc")
set(txt "/* generated by CMakeLists.txt */\n#include \"EST_unix.h\"\n#include \"stdio.h\"\n#include \"festival.h\"\n\n")
foreach(m ${MODULES})
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/modules/${m}/NoInit")
    string(APPEND txt "void festival_${m}_init(void);\n")
  endif()
endforeach()
string(APPEND txt "\nvoid festival_init_modules(void)\n{\n")
foreach(m ${MODULES})
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/modules/${m}/NoInit")
    string(APPEND txt "   festival_${m}_init();\n")
  endif()
endforeach()
string(APPEND txt "}\n")
file(WRITE "${INIT_MODULES}" "${txt}")
list(APPEND FEST_SRCS "${INIT_MODULES}")

add_library(Festival STATIC ${FEST_SRCS})
target_include_directories(Festival PUBLIC
  src/include
  ${EST_DIR}/include)
target_include_directories(Festival PRIVATE
  src/arch/festival
  src/modules/UniSyn
  src/modules/hts_engine
  src/modules/MultiSyn)
target_compile_definitions(Festival PUBLIC SYSTEM_IS_WIN32)
target_compile_definitions(Festival PRIVATE
  FESTIVAL
  NO_SPOOLER
  "FTNAME=Festival Speech Synthesis System"
  FTVERSION=2.5.1
  FTSTATE=current
  "FTDATE=July 2020"
  FTOSTYPEC=x86_64_windows
  FTLIBDIRC=festival/lib)
set_property(TARGET Festival PROPERTY CXX_STANDARD 11)
if(MSVC)
  target_compile_options(Festival PRIVATE /W0 /EHsc)
  target_compile_definitions(Festival PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
else()
  set_property(TARGET Festival PROPERTY CXX_EXTENSIONS ON)
  target_compile_options(Festival PRIVATE -w)
endif()
